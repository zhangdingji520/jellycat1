async function startSearch() { const apiKey = apiKeyInput.value.trim(); if (!apiKey) { alert('🐣 请先填写 Serper API 密钥！'); showError('🐣 请先填写 Serper API 密钥～'); return; } const target = getValidTarget(); if (currentTask) { shouldStop = true; if (resumeResolver) resumeResolver(); await currentTask; } shouldStop = false; isPaused = false; if (resumeResolver) resumeResolver = null; const countries = getSelectedCountries(); if (countries.length === 0) { showError('🌍 至少选择一个国家嗷'); return; } const keywords = buildKeywords(); if (keywords.length === 0) { showError('🔍 请输入至少一个搜索关键词、SKU 或 UPC'); return; } const proxyUrl = proxyUrlInput.value.trim(); if (!proxyUrl) { showError('🖥️ 请填写后端代理地址'); return; } startBtn.disabled = true; pauseBtn.disabled = false; resumeBtn.disabled = true; detectBtn.disabled = true; saveUrlsBtn.disabled = true; errorDiv.style.display = 'none'; resultArea.style.display = 'none'; logArea.style.display = 'block'; clearLog(); addLog(`✨ 开始寻宝之旅，目标 ${target} 个独立站 ✨`); let allResults = []; let totalRequests = 0; let stopCollect = false; currentUsdExcluded = 0; // 翻页配置：减少到 2 页 const PAGES = 2; const RESULTS_PER_PAGE = 20; try { for (const country of countries) { if (stopCollect || shouldStop) break; for (const kw of keywords) { if (stopCollect || shouldStop) break; for (let page = 0; page < PAGES; page++) { if (stopCollect || shouldStop) break; await waitIfPaused(); if (stopCollect || shouldStop) break; const start = page * RESULTS_PER_PAGE; addLog(`🔎 正在搜索 ${country} - “${kw}” (第${page+1}页, start=${start}) ...`); totalRequests++; try { const response = await fetch(proxyUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ query: kw, country, apiKey, start, num: RESULTS_PER_PAGE }) }); if (!response.ok) { const errText = await response.text(); addLog(`⚠️ 请求失败 (${response.status})，跳过此页`, true); continue; } const data = await response.json(); if (data.results && Array.isArray(data.results)) { let filteredBlacklist = 0; let usdThisBatch = 0; let filteredNotProduct = 0; const domainMap = new Map(); for (const item of data.results) { if (isExcludedUrl(item.url)) { filteredBlacklist++; continue; } if (!isLikelyProductUrl(item.url)) { filteredNotProduct++; continue; } if (item.status === 'usd_excluded') { usdThisBatch++; currentUsdExcluded++; continue; } try { const host = new URL(item.url).hostname.replace(/^www\./, ''); if (!domainMap.has(host)) domainMap.set(host, { url: item.url, status: item.status }); } catch(e) {} } const newItems = Array.from(domainMap.values()); allResults.push(...newItems); if (allResults.length > target) allResults = allResults.slice(0, target); collectedResults = [...allResults]; addLog(`📦 收到 ${data.results.length} 个原始结果，过滤黑名单 ${filteredBlacklist} 个，过滤非产品页 ${filteredNotProduct} 个，排除美元站 ${usdThisBatch} 个，剩余有效 ${newItems.length} 个独立域名，累计 ${allResults.length}/${target}`); if (allResults.length >= target) { addLog(`🎯 已达到目标数量 ${target}，停止搜索`); stopCollect = true; break; } } else { addLog(`⚠️ 未获得有效结果`, true); } } catch (err) { addLog(`❌ 请求异常: ${err.message}，跳过此页`, true); continue; } await new Promise(r => setTimeout(r, 200)); } } } } catch (err) { addLog(`❌ 致命错误: ${err.message}`, true); showError(`请求失败: ${err.message}`); startBtn.disabled = false; pauseBtn.disabled = true; resumeBtn.disabled = true; detectBtn.disabled = false; saveUrlsBtn.disabled = true; currentTask = null; return; } collectedResults = allResults; if (shouldStop) { addLog("任务已停止", true); startBtn.disabled = false; pauseBtn.disabled = true; resumeBtn.disabled = true; detectBtn.disabled = false; saveUrlsBtn.disabled = true; currentTask = null; return; } if (allResults.length < target) { addLog(`🏁 搜索结束，仅收集到 ${allResults.length} 个独立域名（未达到目标 ${target}）。`); addLog(`💡 提示：请增加更多搜索关键词、SKU 或 UPC 码，或勾选更多国家，然后重新开始搜索。`); } else { addLog(`🏁 搜索结束，共收集 ${allResults.length} 个独立域名，已达到目标 ${target}。`); } addLog(`请点击「检测库存」查看结果。`); detectBtn.disabled = false; startBtn.disabled = false; pauseBtn.disabled = true; resumeBtn.disabled = true; currentTask = null; }
